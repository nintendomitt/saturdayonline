/**
 * Saturday Online — PayTR ödeme Worker'ı (Cloudflare Workers)
 *
 * Uç noktalar
 *   POST /token            Siteden çağrılır. Paket + müşteri bilgisiyle PayTR iFrame token'ı alır.
 *   POST /paytr/bildirim   PayTR'nin Bildirim URL'si. Hash'i doğrular, siparişi kaydeder, "OK" döner.
 *   GET  /saglik           Basit çalışma kontrolü.
 *
 * Ortam değişkenleri (Cloudflare > Worker > Settings > Variables and Secrets)
 *   MERCHANT_ID    (Text)    PayTR Mağaza No
 *   MERCHANT_KEY   (Secret)  PayTR Mağaza Parolası   — kullanıcı kendisi girer
 *   MERCHANT_SALT  (Secret)  PayTR Mağaza Gizli Anahtarı — kullanıcı kendisi girer
 *   TEST_MODE      (Text)    "1" test, "0" canlı
 *   NOTIFY_URL     (Text, isteğe bağlı) Başarılı ödemede JSON POST atılacak adres (ör. Google Sheets web uygulaması)
 * Bağlantı (Bindings)
 *   ORDERS         (KV namespace, önerilir) Sipariş kaydı ve tekrar gelen bildirimleri ayıklamak için
 */

// Fiyatlar KDV HARİÇ, TL. KDV burada eklenir. Fiyat SADECE burada belirlenir; tarayıcıdan gelen tutara güvenilmez.
const KDV_ORANI = 0.20;
const PAKETLER = {
  'sosyal-medya':  { ad: 'Sosyal Medya Yönetimi (1 ay)', net: 15000, cekimEklenebilir: true },
  'meta-reklam':   { ad: 'Meta Reklam Yönetimi (1 ay)', net: 15000, cekimEklenebilir: true },
  'sm-meta-paket': { ad: 'Sosyal Medya + Meta Reklam Paketi (1 ay, 1 gün çekim dahil)', net: 25000, cekimEklenebilir: false },
};
const CEKIM = { ad: 'Fotoğraf/Video Çekimi (tek gün)', net: 10000 };
const kdvli = net => Math.round(net * (1 + KDV_ORANI));

const SITE = 'https://saturdayonline.co';
const IZINLI_KOKENLER = [SITE, 'https://www.saturdayonline.co'];

export default {
  async fetch(request, env, ctx) {
    const url = new URL(request.url);
    const yol = url.pathname.replace(/\/+$/, '');

    if (request.method === 'OPTIONS') return cors(request, new Response(null, { status: 204 }));

    try {
      if (yol === '/token' && request.method === 'POST') return cors(request, await tokenAl(request, env));
      if (yol === '/paytr/bildirim' && request.method === 'POST') return await bildirim(request, env, ctx);
      if (yol === '/saglik') {
        return json({ ok: true, test_mode: env.TEST_MODE === '1', kv: !!env.ORDERS,
          ayarlar_tamam: !!(env.MERCHANT_ID && env.MERCHANT_KEY && env.MERCHANT_SALT) });
      }
      return new Response('Not found', { status: 404 });
    } catch (e) {
      console.error('Beklenmeyen hata', e && e.stack || e);
      return cors(request, json({ ok: false, hata: 'Sunucu hatası, lütfen tekrar deneyin.' }, 500));
    }
  },
};

/* ───────────── 1) Token ───────────── */

async function tokenAl(request, env) {
  if (!env.MERCHANT_ID || !env.MERCHANT_KEY || !env.MERCHANT_SALT) {
    return json({ ok: false, hata: 'Ödeme altyapısı henüz yapılandırılmadı.' }, 503);
  }
  const koken = request.headers.get('Origin') || '';
  if (!IZINLI_KOKENLER.includes(koken)) return json({ ok: false, hata: 'İzin verilmeyen kaynak.' }, 403);

  let g;
  try { g = await request.json(); } catch { return json({ ok: false, hata: 'Geçersiz istek.' }, 400); }

  const paket = PAKETLER[g.paket];
  if (!paket) return json({ ok: false, hata: 'Paket bulunamadı.' }, 400);

  const ad = temiz(g.ad, 60), email = temiz(g.email, 100), tel = temiz(g.telefon, 20).replace(/[^\d+]/g, '');
  const adres = temiz(g.adres, 400), firma = temiz(g.firma, 120), vergi = temiz(g.vergi, 60);
  if (ad.length < 3) return json({ ok: false, hata: 'Ad soyad girin.' }, 400);
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) return json({ ok: false, hata: 'Geçerli bir e-posta girin.' }, 400);
  if (tel.replace(/\D/g, '').length < 10) return json({ ok: false, hata: 'Geçerli bir telefon girin.' }, 400);
  if (adres.length < 10) return json({ ok: false, hata: 'Fatura adresini girin.' }, 400);
  if (g.sozlesme !== true) return json({ ok: false, hata: 'Sözleşmeleri onaylamanız gerekiyor.' }, 400);

  const cekim = g.cekim === true && paket.cekimEklenebilir;
  const sepet = [[paket.ad, kdvli(paket.net).toFixed(2), 1]];
  if (cekim) sepet.push([CEKIM.ad, kdvli(CEKIM.net).toFixed(2), 1]);
  const toplamTL = kdvli(paket.net) + (cekim ? kdvli(CEKIM.net) : 0);
  const merchant_oid = 'SO' + Date.now() + rastgele(6);          // PayTR: yalnızca harf/rakam
  const payment_amount = String(toplamTL * 100);                    // kuruş, KDV dahil
  const user_basket = btoaUtf8(JSON.stringify(sepet));
  const user_ip = request.headers.get('CF-Connecting-IP') || '';
  const no_installment = '0', max_installment = '0', currency = 'TL';
  const test_mode = env.TEST_MODE === '1' ? '1' : '0';

  const hash_str = env.MERCHANT_ID + user_ip + merchant_oid + email + payment_amount + user_basket
    + no_installment + max_installment + currency + test_mode;
  const paytr_token = await hmacB64(env.MERCHANT_KEY, hash_str + env.MERCHANT_SALT);

  const form = new URLSearchParams({
    merchant_id: env.MERCHANT_ID, user_ip, merchant_oid, email, payment_amount, paytr_token, user_basket,
    debug_on: test_mode, no_installment, max_installment,
    user_name: ad, user_address: (firma ? firma + ' - ' : '') + adres, user_phone: tel,
    merchant_ok_url: SITE + '/odeme/basarili/?s=' + merchant_oid + '&p=' + encodeURIComponent(g.paket) + (cekim ? '&c=1' : ''),
    merchant_fail_url: SITE + '/odeme/basarisiz/?s=' + merchant_oid,
    timeout_limit: '30', currency, test_mode, lang: 'tr',
  });

  const r = await fetch('https://www.paytr.com/odeme/api/get-token', { method: 'POST', body: form });
  const sonuc = await r.json().catch(() => ({}));
  if (sonuc.status !== 'success') {
    console.error('PayTR get-token hatası', sonuc.reason || r.status);
    return json({ ok: false, hata: 'Ödeme başlatılamadı. Lütfen WhatsApp\'tan bize yazın.' }, 502);
  }

  if (env.ORDERS) {
    await env.ORDERS.put('siparis:' + merchant_oid, JSON.stringify({
      merchant_oid, paket: g.paket, paket_ad: paket.ad, cekim, net_tl: paket.net + (cekim ? CEKIM.net : 0), tutar_kurus: Number(payment_amount),
      ad, email, telefon: tel, adres, firma, vergi, durum: 'bekliyor', olusturma: new Date().toISOString(),
      test: test_mode === '1',
    }), { expirationTtl: 60 * 60 * 24 * 400 });
  }
  return json({ ok: true, token: sonuc.token, siparis: merchant_oid });
}

/* ───────────── 2) Bildirim (PayTR → Worker) ───────────── */

async function bildirim(request, env, ctx) {
  const p = Object.fromEntries(new URLSearchParams(await request.text()));
  const beklenen = await hmacB64(env.MERCHANT_KEY || '', (p.merchant_oid || '') + (env.MERCHANT_SALT || '') + (p.status || '') + (p.total_amount || ''));
  if (!p.hash || !esit(beklenen, p.hash)) {
    console.error('PayTR bildirim: hash uyuşmadı', p.merchant_oid);
    return new Response('PAYTR notification failed: bad hash', { status: 400 });
  }

  // PayTR aynı ödeme için birden fazla bildirim gönderebilir; yalnızca ilki işlenir.
  if (env.ORDERS) {
    const anahtar = 'siparis:' + p.merchant_oid;
    const kayit = JSON.parse((await env.ORDERS.get(anahtar)) || 'null') || { merchant_oid: p.merchant_oid };
    if (kayit.durum === 'odendi' || kayit.durum === 'basarisiz') return ok();

    if (p.status === 'success') {
      if (kayit.tutar_kurus && Number(p.payment_amount) !== kayit.tutar_kurus) {
        console.error('Tutar uyuşmazlığı', p.merchant_oid, p.payment_amount, kayit.tutar_kurus);
        kayit.not = 'TUTAR UYUŞMAZLIĞI: PayTR ' + p.payment_amount + ' / beklenen ' + kayit.tutar_kurus;
      }
      Object.assign(kayit, { durum: 'odendi', tahsil_kurus: Number(p.total_amount), taksit: p.installment_count || '1',
        odeme_tipi: p.payment_type || '', odeme_zamani: new Date().toISOString(), test: p.test_mode === '1' });
    } else {
      Object.assign(kayit, { durum: 'basarisiz', hata_kodu: p.failed_reason_code || '', hata: p.failed_reason_msg || '',
        guncelleme: new Date().toISOString() });
    }
    await env.ORDERS.put(anahtar, JSON.stringify(kayit), { expirationTtl: 60 * 60 * 24 * 400 });
    if (kayit.durum === 'odendi' && env.NOTIFY_URL) {
      ctx.waitUntil(fetch(env.NOTIFY_URL, { method: 'POST', headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ tur: 'paytr_odeme', ...kayit }) }).catch(e => console.error('NOTIFY_URL hatası', e)));
    }
  }
  return ok();
}

/* ───────────── Yardımcılar ───────────── */

function ok() { return new Response('OK', { headers: { 'Content-Type': 'text/plain' } }); }

function json(o, status = 200) {
  return new Response(JSON.stringify(o), { status, headers: { 'Content-Type': 'application/json; charset=utf-8' } });
}

function cors(request, res) {
  const koken = request.headers.get('Origin') || '';
  const h = new Headers(res.headers);
  if (IZINLI_KOKENLER.includes(koken)) {
    h.set('Access-Control-Allow-Origin', koken);
    h.set('Access-Control-Allow-Methods', 'POST, OPTIONS');
    h.set('Access-Control-Allow-Headers', 'Content-Type');
    h.set('Vary', 'Origin');
  }
  return new Response(res.body, { status: res.status, headers: h });
}

function temiz(v, max) { return String(v == null ? '' : v).replace(/[\u0000-\u001f<>]/g, ' ').trim().slice(0, max); }

function rastgele(n) {
  const a = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789', b = crypto.getRandomValues(new Uint8Array(n));
  return Array.from(b, x => a[x % a.length]).join('');
}

function btoaUtf8(s) {
  const bytes = new TextEncoder().encode(s); let bin = '';
  for (const x of bytes) bin += String.fromCharCode(x);
  return btoa(bin);
}

async function hmacB64(key, msg) {
  const k = await crypto.subtle.importKey('raw', new TextEncoder().encode(key), { name: 'HMAC', hash: 'SHA-256' }, false, ['sign']);
  const sig = new Uint8Array(await crypto.subtle.sign('HMAC', k, new TextEncoder().encode(msg)));
  let bin = ''; for (const x of sig) bin += String.fromCharCode(x);
  return btoa(bin);
}

function esit(a, b) {
  if (a.length !== b.length) return false;
  let d = 0; for (let i = 0; i < a.length; i++) d |= a.charCodeAt(i) ^ b.charCodeAt(i);
  return d === 0;
}
